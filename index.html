V18 Alarm Patch

คัดลอกสคริปต์ด้านล่างไปไว้ก่อน </body> ของ index.html

const alarmAudio=new Audio("./alarm.wav");
alarmAudio.preload="auto";alarmAudio.volume=1;
async function unlockAudio(){try{await alarmAudio.play();alarmAudio.pause();alarmAudio.currentTime=0;}catch(e){}}
document.addEventListener("touchstart",unlockAudio,{once:true});
document.addEventListener("click",unlockAudio,{once:true});
async function playAlarmSound(){for(let i=0;i<3;i++){try{alarmAudio.currentTime=0;await alarmAudio.play();}catch(e){}await new Promise(r=>setTimeout(r,450));}if(navigator.vibrate)navigator.vibrate([300,150,300]);}
